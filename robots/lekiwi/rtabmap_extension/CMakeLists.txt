cmake_minimum_required(VERSION 3.20)
project(lekiwi_localization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(RTABMap REQUIRED COMPONENTS core)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
pybind11_add_module(lekiwi_localization localize.cpp)
target_include_directories(lekiwi_localization PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lekiwi_localization PRIVATE rtabmap::core ${OpenCV_LIBS})
install(TARGETS lekiwi_localization LIBRARY DESTINATION .)
